    def keypress(self, widget, event):

                self.text = self.txtbox.get_text().lower()
            except:
                pass


            self.txtbox.set_text("")

                if self.antagonist.lower() == self.text:

                    #self.label = self.label.override_color(Gtk.StateFlags.NORMAL, Gdk.RGBA(0, 0, 0, 0))
                    #self.label = self.builder.get_object("word")
                    #auxlabel = self.label.get_text()
                    #self.label = self.label.set_text(auxlabel)
                    GLib.timeout_add(150, self.successmade)
                    self.wordgenerator()
                    self.record += 1
                    self.recordupdate()
                elif self.antagonist.lower() != self.text and self.text != "":
                    self.record = -1
                    GLib.timeout_add(50, self.failmade)
                    self.recordupdate()
#If there's more than one translation is a list, so it searchs inside it.
            elif isinstance(self.antagonist, list):
                self.auxrecord = self.record
                for element in self.antagonist:
                    if element.lower() == self.text:
                        GLib.timeout_add(150, self.successmade)
                        self.wordgenerator()
                        self.record += 1
                        self.recordupdate()
                if self.auxrecord == self.record and self.text != "":
                    self.record = -1
                    GLib.timeout_add(50, self.failmade)
                    self.recordupdate()
            #This conditions are the given ones in case a wrong answers is given and enter is pressed two times.
            if self.record == 0 and self.switch.get_active() == True and self.text == "":
                self.switch.set_active(False)
#Function who recieves a feedback of the result of the answer and acts on it.
    def recordupdate(self):
        self.recordtxtbox = self.builder.get_object("rcrd")
        #Since it starts from 0 it needs a push because record += record don't works with 0.
        if self.record == 0:
            self.recordtxtbox = self.recordtxtbox.set_text("Record: 1 ")
            self.record = 1
        #Minus one is a way of telling the function there was a mistake.
        elif self.record == -1:
            self.recordtxtbox = self.recordtxtbox.set_text("Combo break!")
            #Once a mistake is made the answer is atomatically given since it's expected the user wants it.
            self.switch.set_active(True)
            #self.txtbox = self.builder.get_object("textinput")
        #This else assumes there were no mistakes made and register the increments
        #The increments are registered by the function keypress in case of a right answer is given.
        else:
            self.recordtxtbox = self.recordtxtbox.set_text(r"Record: "+str(self.record)+" ")
            #Here the top score is recorded if it applies.
            if self.hscore < self.record:
                self.toprecord = self.builder.get_object("hscore")
                self.toprecord = self.toprecord.set_text("Top: "+str(self.record))
                self.hscore = self.record
